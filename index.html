Full sketch for 1 frame with a 2 second delay between frames.

#include <FastLED.h>

const uint8_t NUM_X = 8;
const uint8_t NUM_Y = 8;
const uint8_t DATA_PIN = 3;

const uint16_t NUM_LEDS = NUM_X * NUM_Y;
const uint16_t FRAME_DELAY_MS = 2000;

CRGB leds[NUM_LEDS];

CRGB frame1[NUM_LEDS] =
{
  CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),
  CRGB(0,0,0),CRGB(192,0,0),CRGB(192,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(192,0,0),CRGB(192,0,0),CRGB(0,0,0),
  CRGB(192,0,0),CRGB(246,187,0),CRGB(246,187,0),CRGB(192,0,0),CRGB(192,0,0),CRGB(246,187,0),CRGB(246,187,0),CRGB(192,0,0),
  CRGB(192,0,0),CRGB(246,187,0),CRGB(66,135,245),CRGB(66,135,245),CRGB(66,135,245),CRGB(66,135,245),CRGB(246,187,0),CRGB(192,0,0),
  CRGB(0,0,0),CRGB(192,0,0),CRGB(66,135,245),CRGB(66,135,245),CRGB(66,135,245),CRGB(66,135,245),CRGB(192,0,0),CRGB(0,0,0),
  CRGB(0,0,0),CRGB(0,0,0),CRGB(192,0,0),CRGB(49,87,155),CRGB(49,87,155),CRGB(192,0,0),CRGB(0,0,0),CRGB(0,0,0),
  CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(192,0,0),CRGB(192,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),
  CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),CRGB(0,0,0),
};

CRGB* animationFrames[] = { frame1 };
const uint8_t FRAME_COUNT = sizeof(animationFrames) / sizeof(animationFrames[0]);

void setup()
{
  FastLED.addLeds<WS2812B, DATA_PIN, GRB>(leds, NUM_LEDS);
  FastLED.setBrightness(80);
}

void loop()
{
  for (uint8_t frameIndex = 0; frameIndex < FRAME_COUNT; frameIndex++) {
    displayBitmapColors(leds, animationFrames[frameIndex], 8, 8);
    delay(FRAME_DELAY_MS);
  }
}

void displayBitmapColors(CRGB array[], CRGB bitmap[], int rows, int cols) {
  for (int i = 0; i < rows; i++) {
    for (int j = 0; j < cols; j++) {
      array[i * cols + j] = bitmap[i * cols + j];
    }
  }
  FastLED.show();
}
